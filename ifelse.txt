lex.l:-

%{
#include "y.tab.h"
#include <string.h>
extern YYSTYPE yylval;
%}

%%
"if"                    { return IF; }
"else"                  { return ELSE; }
[a-zA-Z][a-zA-Z0-9]*    { yylval.str = strdup(yytext); return ID; }
[0-9]+                  { yylval.str = strdup(yytext); return NUM; }
"<"                     { return LT; }
">"                     { return GT; }
"="                     { return ASSIGN; }
"+"                     { return PLUS; }
"*"                     { return MUL; }
"{"                     { return '{'; }
"}"                     { return '}'; }
"("                     { return '('; }
")"                     { return ')'; }
";"                     { return ';'; }
[ \t\n\r]+              { /* Skip whitespace */ }
%%

int yywrap() { return 1; }


parser.y:-

%{
#include <stdio.h>
#include <stdlib.h>

int yylex(void);
void yyerror(const char *s);
%}

%union {
char *str;
}

%token <str> ID NUM
%token IF ELSE LT GT ASSIGN PLUS MUL

%%

program:
if_stmt {
printf("\nTAC Generated Successfully for If-Else!\n");
}
;

if_stmt:
IF '(' ID LT ID ')' {
printf("t1 = %s < %s\n", $3, $5);
printf("ifFalse t1 goto L1\n");
}
'{' body '}' {
printf("goto L2\n");
printf("L1:\n");
}
ELSE '{' body '}' {
printf("L2:\n");
}
;

body:
body stmt
| stmt
;

stmt:
ID ASSIGN NUM ';' {
printf("%s = %s\n", $1, $3);
}
| ID ASSIGN ID PLUS NUM ';' {
printf("t2 = %s + %s\n", $3, $5);
printf("%s = t2\n", $1);
}
;

%%

void yyerror(const char *s) {
fprintf(stderr, "Error: %s\n", s);
}

int main(void) {
return yyparse();
}


input:-

if (a < b)
{
    x = 5;
}
else
{
    x = 10;
}

commands:-


bison -dy parser.y
flex lex.l
gcc y.tab.c lex.yy.c -o compiler_if
./compiler_if < input.txt
