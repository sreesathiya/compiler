lex.l:-

%{
#include "y.tab.h"
#include <string.h>
extern YYSTYPE yylval;
%}

%%
"while"                 { return WHILE; }
[a-zA-Z][a-zA-Z0-9]*    { yylval.str = strdup(yytext); return ID; }
[0-9]+                  { yylval.str = strdup(yytext); return NUM; }
"<"                     { return LT; }
"="                     { return ASSIGN; }
"+"                     { return PLUS; }
"*"                     { return MUL; }
"{"                     { return '{'; }
"}"                     { return '}'; }
"("                     { return '('; }
")"                     { return ')'; }
";"                     { return ';'; }
[ \t\n\r]+              { /* Skip whitespace */ }
%%

int yywrap() { return 1; }

parser.y:-

%{
#include <stdio.h>
#include <stdlib.h>
int yylex();
void yyerror(const char *s);
%}

%union {
    char* str;
}

%token <str> ID NUM
%token WHILE LT ASSIGN PLUS MUL

%%

program:
    while_stmt {
        printf("\nTAC Generated Successfully!\n");
    }
    ;

while_stmt:
    WHILE '(' ID LT ID MUL NUM ')' {
        // --- 1. Print loop start and condition check FIRST ---
        printf("L1:\n");
        printf("  t1 = %s * %s\n", $5, $7);  // n * 4
        printf("  t2 = %s < t1\n", $3);       // m < t1
        printf("  ifFalse t2 goto L2\n");
    } 
    '{' body '}' {
        // --- 2. Print jump back and loop exit AFTER the body ---
        printf("  goto L1\n");
        printf("L2:\n");
    }
    ;

body:
    body stmt
    | stmt
    ;

stmt:
    ID ASSIGN NUM MUL NUM ';' {
        printf("  t3 = %s * %s\n", $3, $5);
        printf("  %s = t3\n", $1);
    }
    | ID ASSIGN ID MUL ID PLUS ID ';' {
        printf("  t4 = %s * %s\n", $3, $5);
        printf("  t5 = t4 + %s\n", $7);
        printf("  %s = t5\n", $1);
    }
    | ID ASSIGN ID PLUS ID ';' {
        printf("  t6 = %s + %s\n", $3, $5);
        printf("  %s = t6\n", $1);
    }
    ;

%%

void yyerror(const char *s) {
    fprintf(stderr, "Error: %s\n", s);
}

int main() {
    yyparse();
    return 0;
}

input.txt:-
while (m < n * 4)
{
    r = 16 * 5;
    s = x * y + z;
    t = m + s;
}


command:

bison -dy parser.y
flex lex.l
gcc y.tab.c lex.yy.c -o compiler_if
./compiler_if < input.txt
